#include<stdio.h>
int main()
{
    int marks;
    printf("enter marks:");
    scanf("%d",&marks);
    if(marks>=90)
    {
        printf("grade:A");
    }
    else if(marks>=70)
    {
        printf("grade:B");
    }
    else if(marks>=55)
    {
        printf("grade:C");
    }
    else if(marks>=45)
    {
        printf("grade:D");
    }
    else if(marks>=37)
    {
        printf("grade:E");
    }
    else
    {
        printf("grade:F");
    }
}
